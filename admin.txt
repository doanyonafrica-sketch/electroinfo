function cleanQuillHTML(html) {
    return html
        .replace(/class="ql-[^"]*"/g, '')
        .replace(/<p>\s*<br\s*\/?>\s*<\/p>/gi, '')
        .replace(/<p>\s*&nbsp;\s*<\/p>/gi, '')
        .replace(/<p>\s*<\/p>/gi, '')
        .replace(/<br\s*\/?>\s*<\/p>/gi, '</p>')
        .replace(/(<br\s*\/?>){2,}/gi, '<br>')
        .replace(/\s+/g, ' ')
        .replace(/>\s+</g, '><')
        .trim();
}

function getContentSize(content) {
    return new TextEncoder().encode(content).length;
}

function truncateContent(html, maxBytes = 950000) {
    const encoder = new TextEncoder();
    const encoded = encoder.encode(html);
    if (encoded.length <= maxBytes) return html;
    console.warn(`⚠️ Contenu trop long: ${encoded.length} bytes.`);
    let truncated = html.substring(0, Math.floor(html.length * maxBytes / encoded.length));
    const lastTag   = truncated.lastIndexOf('<');
    const lastClose = truncated.lastIndexOf('>');
    if (lastTag > lastClose) truncated = truncated.substring(0, lastTag);
    return truncated + '\n<p><em style="color:#e74c3c;">[⚠️ Contenu tronqué — article trop long]</em></p>';
}

function generateSlug(title) {
    return title
        .toLowerCase()
        .normalize('NFD')
        .replace(/[\u0300-\u036f]/g, '')
        .replace(/[^a-z0-9\s-]/g, '')
        .trim()
        .replace(/\s+/g, '-')
        .replace(/-+/g, '-')
        .substring(0, 100);
}

async function isSlugUnique(slug, excludeId = null) {
    const snapshot = await getDocs(query(collection(db, 'articles'), where('slug', '==', slug)));
    if (snapshot.empty) return true;
    if (excludeId) return snapshot.docs[0].id === excludeId;
    return false;
}

async function generateUniqueSlug(title, excludeId = null) {
    const base = generateSlug(title);
    let slug = base, counter = 1;
    while (!(await isSlugUnique(slug, excludeId))) slug = `${base}-${counter++}`;
    return slug;
}

async function compressImage(file, maxWidth = 1200, quality = 0.85) {
    return new Promise((resolve) => {
        const reader = new FileReader();
        reader.onload = (e) => {
            const img = new Image();
            img.onload = () => {
                const canvas = document.createElement('canvas');
                let w = img.width, h = img.height;
                if (w > maxWidth) { h = (h * maxWidth) / w; w = maxWidth; }
                canvas.width = w; canvas.height = h;
                canvas.getContext('2d').drawImage(img, 0, 0, w, h);
                const compressed = canvas.toDataURL('image/jpeg', quality);
                console.log(`📸 ${w}x${h}, ${Math.round((compressed.length*3)/4/1024)} KB`);
                resolve(compressed);
            };
            img.src = e.target.result;
        };
        reader.readAsDataURL(file);
    });
}

function customImageHandler() {
    const input = document.createElement('input');
    input.type = 'file'; input.accept = 'image/*';
    input.onchange = async () => {
        const file = input.files[0];
        if (!file) return;
        const quality = await showImageQualityDialog(file.size / (1024*1024));
        if (quality === null) return;
        showNotification('🔄 Compression...', 'info');
        try {
            const compressed = await compressImage(file, quality.maxWidth, quality.quality);
            const range = quillEditor.getSelection(true);
            quillEditor.insertEmbed(range.index, 'image', compressed);
            quillEditor.setSelection(range.index + 1);
            setTimeout(makeImagesResizable, 100);
            showNotification('✅ Image ajoutée !', 'success');
        } catch (e) {
            showNotification('❌ Erreur image', 'error');
        }
    };
    input.click();
}

}

function insertYouTubeVideo() {
    const url = prompt('URL YouTube :');
    if (!url) return;
    const id = extractYouTubeID(url);
    if (!id) { showNotification('URL YouTube invalide', 'error'); return; }
    const html = `<div class="video-wrapper" style="position:relative;padding-bottom:56.25%;height:0;margin:2rem 0;">
        <iframe src="https://www.youtube.com/embed/${id}?enablejsapi=1" frameborder="0"
            allow="accelerometer;autoplay;clipboard-write;encrypted-media;gyroscope;picture-in-picture"
            allowfullscreen style="position:absolute;top:0;left:0;width:100%;height:100%;"></iframe>
    </div>`;
    const range = quillEditor.getSelection(true);
    quillEditor.clipboard.dangerouslyPasteHTML(range.index, html);
    quillEditor.setSelection(range.index + 1);
    showNotification('Vidéo ajoutée !', 'success');
}

function extractYouTubeID(url) {
    let m;
    m = url.match(/[?&]v=([^&]+)/); if (m) return m[1];
    m = url.match(/youtu\.be\/([^?]+)/); if (m) return m[1];
    m = url.match(/youtube\.com\/embed\/([^?]+)/); if (m) return m[1];
    if (url.length === 11 && /^[a-zA-Z0-9_-]+$/.test(url)) return url;
    return null;
}

                    `Limite Firebase : 1MB.\n\nOK = tronquer / Annuler = réduire manuellement`
                );

                content = truncateContent(content);
            }

            const textContent = quillEditor.getText().trim();
            if (!textContent) {

            }

        const category = document.getElementById('category').value;

        if (!title)   { showNotification('⚠️ Le titre est obligatoire', 'error'); return; }

            title, slug, category,

            contentType: editorMode,         // ✅ type d'éditeur sauvegardé

        const totalSize = getContentSize(JSON.stringify(articleData));

            showNotification('❌ Article trop volumineux', 'error');

            return;
        }

        if (editMode && currentEditId) {

        loadStatistics();

    } catch (error) {
        console.error('❌ Erreur publication:', error);
        let msg = 'Erreur lors de la publication';
        if (error.code === 'invalid-argument' && error.message.includes('longer than'))
            msg = 'Article trop volumineux. Réduisez le contenu ou les images.';
        showNotification(`❌ ${msg}: ${error.message}`, 'error');
    }

// ============================================
async function loadArticles() {
    try {
        ['publishedArticlesList','scheduledArticlesList','draftsArticlesList'].forEach(id => {
            document.getElementById(id).innerHTML = '<div class="loading"><i class="fas fa-spinner fa-spin"></i><p>Chargement...</p></div>';
        });

        const snapshot = await getDocs(query(collection(db, 'articles'), orderBy('createdAt', 'desc')));
        const published = [], scheduled = [], drafts = [];

        snapshot.forEach(d => {
            const status = d.data().status || 'published';
            const item   = { id: d.id, data: d.data() };
            if (status === 'published') published.push(item);
            else if (status === 'scheduled') scheduled.push(item);
            else if (status === 'draft') drafts.push(item);
        });

        document.getElementById('publishedCount').textContent = published.length;
        document.getElementById('scheduledCount').textContent = scheduled.length;
        document.getElementById('draftsCount').textContent    = drafts.length;

        if (typeof window.loadArticlesEnhanced === 'function') {
            await window.loadArticlesEnhanced(snapshot); return;
        }

        window.__pendingSnapshot = snapshot;
        window.dispatchEvent(new CustomEvent('articlesLoaded', { detail: snapshot }));

        displayArticlesInSection('publishedArticlesList', published,  'publié');
        displayArticlesInSection('scheduledArticlesList', scheduled,  'programmé');
        displayArticlesInSection('draftsArticlesList',    drafts,     'brouillon');

    } catch (error) {
        console.error('Erreur chargement:', error);
        document.getElementById('publishedArticlesList').innerHTML =
            '<div class="empty-state"><i class="fas fa-exclamation-triangle"></i><p>Erreur de chargement</p></div>';
    }
}

function displayArticlesInSection(sectionId, articles, type) {
    const container = document.getElementById(sectionId);
    if (articles.length === 0) {
        const cfg = {
            'publié':     { icon:'check-circle', msg:'Aucun article publié',    hint:'Publiez votre premier article !' },
            'programmé':  { icon:'clock',        msg:'Aucun article programmé', hint:'Programmez une publication future' },
            'brouillon':  { icon:'file-alt',     msg:'Aucun brouillon',         hint:'Enregistrez vos articles en brouillon' }
        }[type] || { icon:'file', msg:'Aucun article', hint:'' };
        container.innerHTML = `<div class="empty-state"><i class="fas fa-${cfg.icon}"></i><p>${cfg.msg}</p><small>${cfg.hint}</small></div>`;
        return;
    }
    container.innerHTML = '';
    articles.forEach(item => container.appendChild(createArticleItem(item.id, item.data)));
}

export function createArticleItem(id, article) {
    const div  = document.createElement('div');
    div.className = 'admin-article-item';
    const date = article.createdAt
        ? new Date(article.createdAt.toDate()).toLocaleDateString('fr-FR',{day:'numeric',month:'long',year:'numeric'})
        : 'Non daté';
    const featuredBadge = article.featured ? '<span class="badge badge-yellow"><i class="fas fa-star"></i> Vedette</span>' : '';
    const status = article.status || 'published';
    let statusBadge = '';
    if (status === 'draft') {
        statusBadge = '<span class="badge badge-gray"><i class="fas fa-file-alt"></i> Brouillon</span>';
    } else if (status === 'scheduled') {
        const sd = article.scheduledFor ? new Date(article.scheduledFor.toDate()).toLocaleDateString('fr-FR',{day:'numeric',month:'short',hour:'2-digit',minute:'2-digit'}) : '';
        statusBadge = `<span class="badge badge-orange"><i class="fas fa-clock"></i> Programmé ${sd}</span>`;
    } else {
        statusBadge = '<span class="badge badge-green"><i class="fas fa-check-circle"></i> Publié</span>';
    }
    const slugInfo     = article.slug ? `<br><small style="color:#6b7280;"><i class="fas fa-link"></i> /article/${article.slug}</small>` : '';
    const publishNowBtn = status !== 'published'
        ? `<button class="btn btn-sm btn-success" onclick="publishNow('${id}')"><i class="fas fa-rocket"></i> Publier</button>`
        : '';
    div.innerHTML = `
        <div class="article-info">
            <h3>${escapeHtml(article.title)} ${featuredBadge} ${statusBadge}</h3>
            ${slugInfo}
            <p class="article-meta">
                <span class="badge badge-${getCategoryClass(article.category)}">${escapeHtml(article.category)}</span>
                <span>${date}</span>
                <span><i class="fas fa-eye"></i> ${article.views||0}</span>
                <span><i class="fas fa-comment"></i> ${article.commentsCount||0}</span>
            </p>
        </div>
        <div class="article-actions">
            ${publishNowBtn}
            <button class="btn btn-sm btn-success" onclick="generateSharePage('${id}')"><i class="fas fa-share-alt"></i> Partage</button>
            <button class="btn btn-sm btn-primary" onclick="editArticle('${id}')"><i class="fas fa-edit"></i> Modifier</button>
            <button class="btn btn-sm btn-danger"  onclick="showDeleteModal('${id}')"><i class="fas fa-trash"></i> Supprimer</button>
        </div>`;
    return div;
}

// ============================================
window.publishNow = async function(articleId) {
    if (!confirm('Publier cet article immédiatement ?')) return;
    try {
        await updateDoc(doc(db, 'articles', articleId), { status:'published', publishedAt: serverTimestamp(), scheduledFor: null });
        showNotification('✅ Article publié !', 'success');
        loadArticles(); loadStatistics();
    } catch (e) { showNotification('❌ Erreur publication', 'error'); }
};

// ✏️ MODIFIER ARTICLE
// ✅ FIX : charge dans le bon éditeur selon contentType
// ============================================
window.editArticle = async function(articleId) {
    try {
        const docSnap = await getDoc(doc(db, 'articles', articleId));
        if (!docSnap.exists()) { showNotification('Article introuvable', 'error'); return; }
        const article = docSnap.data();

        document.getElementById('title').value    = article.title;
        document.getElementById('category').value = article.category;
        document.getElementById('imageUrl').value = article.imageUrl || '';
        document.getElementById('summary').value  = article.summary;
        document.getElementById('featured').checked = article.featured || false;
        document.getElementById('tags').value     = (article.tags || []).join(', ');

        if (window.fillTranslationFields) window.fillTranslationFields(article);

        const status = article.status || 'published';
        document.getElementById('publicationStatus').value = status;
        if (status === 'scheduled' && article.scheduledFor) {
            const sd = new Date(article.scheduledFor.toDate());
            document.getElementById('scheduleDate').value = sd.toISOString().split('T')[0];
            document.getElementById('scheduleTime').value = sd.toTimeString().slice(0,5);
        }
        if (window.toggleScheduleFields) window.toggleScheduleFields();

        // ✅ Basculer sur le bon éditeur selon contentType sauvegardé
        const savedType = article.contentType || 'quill';
        if (typeof window.switchArticleEditorMode === 'function') {
            window.switchArticleEditorMode(savedType);
        }

        setTimeout(() => {
            if (savedType === 'html') {
                if (window.articleCmInstance) {
                    window.articleCmInstance.setValue(article.content || '');
                    window.articleCmInstance.refresh();
                    // ✅ Synchroniser aussi le champ caché
                    document.getElementById('content').value = article.content || '';
                }
            } else if (savedType === 'wy') {
                const wyBody = document.getElementById('wyBody');
                if (wyBody) {
                    wyBody.innerHTML = article.content || '';
                    document.getElementById('content').value = article.content || '';
                }
            } else {
                if (quillEditor) quillEditor.root.innerHTML = article.content || '';
            }
        }, 150);

        editMode = true;
        currentEditId = articleId;
        document.getElementById('formTitle').textContent    = 'Modifier l\'article';
        document.getElementById('submitBtnText').textContent = 'Mettre à jour';
        window.scrollTo({ top:0, behavior:'smooth' });

    } catch (error) {
        console.error('Erreur chargement article:', error);
        showNotification('Erreur lors du chargement', 'error');
    }
};

window.cancelEdit = function() {
    editMode = false; currentEditId = null;
    document.getElementById('formTitle').textContent     = 'Nouvel Article';
    document.getElementById('submitBtnText').textContent = 'Publier';
    document.getElementById('publicationStatus').value   = 'published';
    if (window.toggleScheduleFields) window.toggleScheduleFields();
    articleForm.reset();
    if (quillEditor) quillEditor.setText('');
    if (window.articleCmInstance) window.articleCmInstance.setValue('');
    const wyBody = document.getElementById('wyBody');
    if (wyBody) wyBody.innerHTML = '';
    document.getElementById('content').value = '';
};

// 🗑️ SUPPRESSION
// ============================================
window.showDeleteModal = function(articleId) {
    articleToDelete = articleId;
    document.getElementById('deleteModal').classList.remove('hidden');
};
document.getElementById('cancelDelete')?.addEventListener('click', () => {
    document.getElementById('deleteModal').classList.add('hidden');
    articleToDelete = null;
});
document.getElementById('confirmDelete')?.addEventListener('click', async () => {
    if (!articleToDelete) return;
    try {
        await deleteDoc(doc(db, 'articles', articleToDelete));
        showNotification('Article supprimé', 'success');
        document.getElementById('deleteModal').classList.add('hidden');
        articleToDelete = null;
        loadArticles(); loadStatistics();
    } catch (e) { showNotification('Erreur suppression', 'error'); }
});

// 📊 STATISTIQUES
// ============================================
async function loadStatistics() {
    try {
        const snap = await getDocs(collection(db, 'articles'));
        let total = snap.size, pub = 0, sched = 0, draft = 0;
        snap.forEach(d => {
            const s = d.data().status || 'published';
            if (s === 'published') pub++;
            else if (s === 'scheduled') sched++;
            else if (s === 'draft') draft++;
        });
        document.getElementById('totalArticles').textContent      = total;
        document.getElementById('publishedArticles').textContent  = pub;
        document.getElementById('scheduledArticles').textContent  = sched;
        document.getElementById('draftArticles').textContent      = draft;
    } catch (e) { console.error('Erreur stats:', e); }
}

// ============================================
async function loadNewsletterSubscribers() {
    try {
        newsletterList.innerHTML = '<div class="loading"><i class="fas fa-spinner fa-spin"></i><p>Chargement...</p></div>';
        const snap = await getDocs(collection(db, 'newsletter'));
        if (snap.empty) { newsletterList.innerHTML = '<p class="empty-text">Aucun abonné</p>'; return; }
        newsletterList.innerHTML = '';
        snap.forEach(d => {
            const sub  = d.data();
            const date = sub.subscribedAt ? new Date(sub.subscribedAt.toDate()).toLocaleDateString('fr-FR') : 'N/A';
            const div  = document.createElement('div');
            div.className = 'newsletter-item';
            div.innerHTML = `<span class="newsletter-email">${escapeHtml(sub.email)}</span><span class="newsletter-date">${date}</span>`;
            newsletterList.appendChild(div);
        });
    } catch (e) { newsletterList.innerHTML = '<p class="empty-text text-danger">Erreur de chargement</p>'; }
}

window.exportNewsletterCSV = async function() {
    try {
        const snap = await getDocs(collection(db, 'newsletter'));
        if (snap.empty) { showNotification('Aucun abonné', 'info'); return; }
        let csv = 'Email,Date\n';
        snap.forEach(d => {
            const sub  = d.data();
            const date = sub.subscribedAt ? new Date(sub.subscribedAt.toDate()).toLocaleDateString('fr-FR') : 'N/A';
            csv += `${sub.email},${date}\n`;
        });
        const a = Object.assign(document.createElement('a'), {
            href: URL.createObjectURL(new Blob([csv], { type:'text/csv' })),
            download: `newsletter-${new Date().toISOString().split('T')[0]}.csv`
        });
        a.click(); URL.revokeObjectURL(a.href);
        showNotification('Export réussi !', 'success');
    } catch (e) { showNotification('Erreur export', 'error'); }
};

// 🔗 PAGE DE PARTAGE
// ============================================
window.generateSharePage = async function(articleId) {
    try {
        showNotification('🔄 Génération...', 'info');
        const snap = await getDoc(doc(db, 'articles', articleId));
        if (!snap.exists()) { showNotification('❌ Article introuvable', 'error'); return; }
        const article = snap.data();
        const slug    = article.slug || articleId;
        const title   = escapeHtml(article.title || 'Article');
        const desc    = escapeHtml(article.summary || 'Découvrez cet article sur Électro-Actu');
        const img     = escapeHtml(article.imageUrl || 'https://electroinfo.online/images/logo.png');
        const shareUrl = `https://electroinfo.online/share/${slug}.html`;
        const html = `<!DOCTYPE html><html lang="fr"><head><meta charset="UTF-8">
<title>${title} | Électro-Actu</title><meta name="description" content="${desc}">
<meta property="og:type" content="article"><meta property="og:title" content="${title}">
<meta property="og:description" content="${desc}"><meta property="og:image" content="${img}">
<meta property="og:url" content="${shareUrl}">
<meta name="twitter:card" content="summary_large_image"><meta name="twitter:title" content="${title}">
<meta name="twitter:image" content="${img}">
<meta http-equiv="refresh" content="0;url=/article/${slug}">
<script>window.location.href="/article/${slug}";<\/script>
<style>body{font-family:sans-serif;display:flex;align-items:center;justify-content:center;min-height:100vh;background:linear-gradient(135deg,#667eea,#764ba2);color:white;text-align:center;}</style>
</head><body><div><h1>${title}</h1><p>Chargement...</p></div></body></html>`;
        const a = Object.assign(document.createElement('a'), {
            href: URL.createObjectURL(new Blob([html], { type:'text/html;charset=utf-8' })),
            download: `${slug}.html`
        });
        document.body.appendChild(a); a.click(); document.body.removeChild(a);
        URL.revokeObjectURL(a.href);
        if (navigator.clipboard) navigator.clipboard.writeText(shareUrl).catch(()=>{});
        showNotification('✅ Fichier téléchargé !', 'success');
        alert(`✅ Page de partage générée !\n\nFichier : ${slug}.html\nURL : ${shareUrl}\n\nCopiez ce fichier dans le dossier /share/ et déployez.`);
    } catch (e) { showNotification('❌ Erreur génération', 'error'); }
};

// 🔄 MIGRATION SLUGS
// ============================================
window.migrateOldArticles = async function() {
    if (!confirm('Ajouter des slugs à tous les articles qui n\'en ont pas ?')) return;
    showNotification('🔄 Migration...', 'info');
    const snap = await getDocs(collection(db, 'articles'));
    let migrated = 0, errors = 0;
    for (const d of snap.docs) {
        if (!d.data().slug) {
            try {
                const slug = await generateUniqueSlug(d.data().title, d.id);
                await updateDoc(doc(db, 'articles', d.id), { slug });
                migrated++;
            } catch (e) { errors++; }
        }
    }
    showNotification(`✅ ${migrated} migrés${errors ? `, ${errors} erreurs` : ''}`, 'success');
    loadArticles();
};

// ============================================
function getCategoryClass(cat) {
    return { INNOVATION:'blue', SÉCURITÉ:'red', NOUVEAUTÉ:'green', TUTO:'orange', DOMOTIQUE:'purple' }[cat] || 'blue';
}

function escapeHtml(text) {
    const d = document.createElement('div');
    d.textContent = text;
    return d.innerHTML;
}

function showNotification(message, type = 'info') {
    const n = document.getElementById('notification');
    n.textContent = message;
    n.className = `notification ${type}`;
    n.classList.remove('hidden');
    setTimeout(() => n.classList.add('hidden'), 3000);
}
